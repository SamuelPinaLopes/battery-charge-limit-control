#include <iostream>
#include <filesystem>
namespace fs = std::filesystem;

int main() {
    fs::path directoryPath = ".";  // Current directory
    
    // STEP 1: Create the iterator
    fs::directory_iterator it(directoryPath);
    // Now 'it' is like a bookmark pointing to the FIRST item in the directory
    
    // STEP 2: Get the first item
    fs::directory_entry entry1 = *it;  // *it gets whatever 'it' is pointing to
    std::cout << entry1.path().filename() << std::endl;
    
    // STEP 3: Move to the second item
    ++it;  // Move the bookmark forward
    fs::directory_entry entry2 = *it;
    std::cout << entry2.path().filename() << std::endl;
    
    // STEP 4: Move to the third item
    ++it;
    fs::directory_entry entry3 = *it;
    std::cout << entry3.path().filename() << std::endl;
    
    return 0;
}
