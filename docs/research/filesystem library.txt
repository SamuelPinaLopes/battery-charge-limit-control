how to use std::filesystem::create_directory correctly in C++:

Basic Usage
cpp
#include <filesystem>
#include <iostream>

namespace fs = std::filesystem;

int main() {
    // Create a single directory
    fs::create_directory("/tmp/myfolder");
    
    return 0;
}
Return Value (Important)
create_directory returns a bool:

true → Directory was created (didn't exist before)

false → Directory already existed

cpp
if (fs::create_directory("/tmp/myfolder")) {
    std::cout << "Directory created" << std::endl;
} else {
    std::cout << "Directory already exists" << std::endl;
}
Create Parent Directories Too
Use create_directories (plural) to create the whole path:

cpp
// This fails if "/tmp/a/b/" doesn't exist
fs::create_directory("/tmp/a/b/c");

// This creates all missing parent directories
fs::create_directories("/tmp/a/b/c");  // ✅ Works
Check Before Creating
cpp
std::string path = "/tmp/myfolder";

if (!fs::exists(path)) {
    if (fs::create_directory(path)) {
        std::cout << "Success" << std::endl;
    } else {
        std::cout << "Failed" << std::endl;
    }
}
Common Errors
Error	Why	Fix
permission denied	No write permission	Use /tmp/ or run as root
file exists	A file (not directory) with same name exists	Check with fs::is_directory()
no space left	Disk full	Free space
Complete Example for Your Boot Detection
cpp
#include <filesystem>
#include <fstream>

namespace fs = std::filesystem;

int main() {
    fs::path dir = "/tmp/myapp";
    fs::path file = dir / "boot_id.txt";
    
    // Create directory if it doesn't exist
    if (!fs::exists(dir)) {
        fs::create_directories(dir);  // Use plural for nested paths
    }
    
    // Write to file
    std::ofstream out(file);
    out << "some data";
    out.close();
    
    return 0;
}
Quick Reference
Function	What it does
fs::create_directory(path)	Creates ONE directory. Fails if parents missing.
fs::create_directories(path)	Creates entire path. Creates parents as needed.
fs::exists(path)	Checks if path exists (file or dir)
fs::is_directory(path)	Checks if path is a directory
fs::remove(path)	Deletes empty directory
fs::remove_all(path)	Deletes directory and all contents
