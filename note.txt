#include <iostream>
#include <filesystem>
namespace fs = std::filesystem;

int main() {
    fs::path directoryPath = ".";  // Current directory
    
    // STEP 1: Create the iterator
    fs::directory_iterator it(directoryPath);
    // Now 'it' is like a bookmark pointing to the FIRST item in the directory
    
    // STEP 2: Get the first item
    fs::directory_entry entry1 = *it;  // *it gets whatever 'it' is pointing to
    std::cout << entry1.path().filename() << std::endl;
    
    // STEP 3: Move to the second item
    ++it;  // Move the bookmark forward
    fs::directory_entry entry2 = *it;
    std::cout << entry2.path().filename() << std::endl;
    
    // STEP 4: Move to the third item
    ++it;
    fs::directory_entry entry3 = *it;
    std::cout << entry3.path().filename() << std::endl;
    
    return 0;
}

but within a for loop and using an iterator:
#include <iostream>
#include <vector>
#include <string>

int main() {
    std::vector<std::string> filenames = {"file1.txt", "file2.txt", "image.jpg"};
    
    // Create iterator
    std::vector<std::string>::iterator it = filenames.begin();
    std::vector<std::string>::iterator end = filenames.end();
    
    for (; it != end; ++it) {
        // *it is the string at current position
        std::string currentFile = *it;  // Dereference to get the string
        
        // Now you can use string methods!
        if (currentFile.compare(0, 4, "file") == 0) {
            std::cout << currentFile << " starts with 'file'" << std::endl;
        }
        
        // Or directly without copying:
        if ((*it).length() > 10) {
            std::cout << *it << " is long" << std::endl;
        }
        
        // Or use arrow operator (more common):
        if (it->find(".txt") != std::string::npos) {
            std::cout << it->c_str() << " is a text file" << std::endl;
        }
    }
    
    return 0;
}
